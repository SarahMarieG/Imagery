#!/bin/bash

PARTDIR="/Volumes/ELEMENTS/imagery/participants"

BETADIR="$OUTDIR/Betas"

for cond in pleasant_imagine neutral_imagine unpleasant_imagine; do
    files=()
    for subdir in "$PARTDIR"/par_*; do
        SUBJ=$(basename "$subdir")

done
