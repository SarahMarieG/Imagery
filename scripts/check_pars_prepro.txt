#!/bin/bash
ROOT="/Volumes/ELEMENTS/imagery/participants"

for subdir in "$ROOT"/par_*; do
  SUBJ="${subdir##*/}"
  f="$subdir/fmri/sswarper/anatQQ.${SUBJ}.nii"
  [[ -f "$f" ]] || continue
  echo "=== $SUBJ ==="
  3dinfo -history "$f" \
    | grep -oE '3d[A-Za-z]+|@[A-Za-z_]+' \
    | tr '\n' ' '
  echo
done
